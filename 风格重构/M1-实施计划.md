# M1 令牌与排版 实施计划（风格重构）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把《风格重构/风格拆解清单.md》（浅色版设计规范）的令牌层与排版层落地到 `frontend/src/style.css`，实现全站"一次换肤"——所有页面无需逐个改动即呈现新的浅色设计语言，并清掉三处最刺眼的旧硬编码色。

**Architecture:** 令牌集中在 `style.css` 的 `:root`（现状已如此），Element Plus / Vant 都通过 CSS 变量覆盖主题（现状已如此），因此 M1 是**纯变量替换 + 别名兼容**：新令牌按规范定义，旧令牌名保留为指向新值的别名，让 35 个页面里已有的 `var(--brand-600)` 等引用自动换肤；后续模块（M2-M6）再逐步把引用迁移到新令牌名并在收尾时删除别名。颜色值由测试锁定（含 WCAG AA 对比度断言），防止后续模块手滑改浅。

**Tech Stack:** Vue 3 + Vite + Element Plus（CSS 变量主题）+ Vant；测试用 vitest（`// @vitest-environment node`，直接读文件文本做断言，不需要 DOM）。

## Global Constraints

以下约束来自《风格重构/风格拆解清单.md》，每个任务的实现都必须满足（值必须逐字一致）：

- **令牌值（浅色版）**：`--canvas:#f1f2f4` `--panel:#fcfcfd` `--panel2:#f5f6f8` `--panel3:#ecedf0`；`--line:rgba(17,19,23,.08)` `--line2:rgba(17,19,23,.14)`；`--txt:#17191e` `--mut:#5b616b` `--dim:#666c75`；`--lime:#cbf74e` `--lime-ink:#15170a` `--lime-text:#5b7600` `--lime-hover:#bceb3a` `--lime-active:#addc28` `--lime-tint:rgba(163,209,26,.14)`；`--success:#17803d` `--warning:#c2410c` `--danger:#c0352b` `--info:#1d4ed8`；阴影一律 `rgba(20,24,32,.05~.22)`；圆角 `--r-frame:32px --r-card:24px --r-in:16px --r-ctl:12px --r-sm:10px`；缓动 `--ease:cubic-bezier(.22,1,.36,1)`
- **禁止**：纯白 `#fff/#ffffff` 作面板底色；纯黑阴影 `rgba(0,0,0,…)`；给任何文字用 `--lime` 原色（文字型强调只能用 `--lime-text`）
- **强制**：正文对比 ≥ 4.5:1（由测试断言）；`--el-font-size-base: 13px`（密度 7 适配）；`--el-font-family: inherit` 保留
- **交互行为不得改变**：全局 toast 的"居中 + 遮罩 + 白底黑字（alert 式）"是本项目刻意的交互设计（见 `style.css:228-274` 注释与 `main.ts` 的 MutationObserver），**只换配色、不动位置与遮罩逻辑**
- **本期不做**：深色模式、⌘K 命令面板、sparkline、看板拖拽、任何新功能（纯样式重构）
- **提交规范**：每个 Task 独立 commit；提交信息用 `style(restyle-m1): …` 前缀
- **分支**：从 `dev` 拉 `feature/restyle-m1-tokens`（见 Task 0）

// @vitest-environment node
// 设计令牌锁定测试：直接读文件文本断言（不依赖 DOM，node 环境最快）
// 规范来源：《风格重构/风格拆解清单.md》§1；颜色值变更必须先改规范再改这里
import { describe, expect, it } from 'vitest'
import { readFileSync } from 'node:fs'

const css = readFileSync(new URL('../style.css', import.meta.url), 'utf8')

function token(name: string): string {
  const m = css.match(new RegExp(`--${name}\\s*:\\s*([^;]+);`))
  if (!m) throw new Error(`缺少令牌 --${name}`)
  return m[1].trim().toLowerCase()
}

function luminance(hex: string): number {
  const h = hex.replace('#', '')
  const [r, g, b] = [0, 2, 4].map((i) => parseInt(h.slice(i, i + 2), 16) / 255)
  const lin = (c: number) => (c <= 0.03928 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4))
  return 0.2126 * lin(r) + 0.7152 * lin(g) + 0.0722 * lin(b)
}

function contrast(a: string, b: string): number {
  const [hi, lo] = [luminance(a), luminance(b)].sort((x, y) => y - x)
  return (hi + 0.05) / (lo + 0.05)
}

describe('设计令牌（浅色版）', () => {
  it('表面与文字令牌为规范值', () => {
    expect(token('canvas')).toBe('#f1f2f4')
    expect(token('panel')).toBe('#fcfcfd')
    expect(token('panel2')).toBe('#f5f6f8')
    expect(token('panel3')).toBe('#ecedf0')
    expect(token('txt')).toBe('#17191e')
    expect(token('mut')).toBe('#5b616b')
    expect(token('dim')).toBe('#666c75')
  })

  it('lime 家族与语义色为规范值', () => {
    expect(token('lime')).toBe('#cbf74e')
    expect(token('lime-ink')).toBe('#15170a')
    expect(token('lime-text')).toBe('#5b7600')
    expect(token('lime-deep')).toBe('#4a6100')
    expect(token('lime-hover')).toBe('#bceb3a')
    expect(token('lime-active')).toBe('#addc28')
    expect(token('success')).toBe('#17803d')
    expect(token('warning')).toBe('#c2410c')
    expect(token('danger')).toBe('#c0352b')
    expect(token('info')).toBe('#1d4ed8')
  })

  it('禁止纯白面板与纯黑阴影', () => {
    expect(token('panel')).not.toMatch(/^#(fff|ffffff)$/)
    for (const name of ['shadow-card', 'shadow-pop', 'shadow-overlay']) {
      expect(token(name)).not.toContain('rgba(0, 0, 0')
      expect(token(name)).not.toContain('rgba(0,0,0')
    }
  })

  it('对比度全部达 WCAG AA', () => {
    const panel = token('panel')
    expect(contrast(token('txt'), panel)).toBeGreaterThanOrEqual(7)
    expect(contrast(token('mut'), panel)).toBeGreaterThanOrEqual(4.5)
    expect(contrast(token('dim'), panel)).toBeGreaterThanOrEqual(4.5)
