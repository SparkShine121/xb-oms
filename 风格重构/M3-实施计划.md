# M3 壳层重构 实施计划（风格重构）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 `MainLayout` 从"通栏侧栏 + 通栏顶栏"重构为规范 §4 的**三面板悬浮壳**（76px 图标 rail + 顶栏面包屑 + 圆角内容区，仅内容区滚动），并把 `Login.vue` 重构为左右分栏的品牌页——观感变化最大的一个模块，同时修掉 M1 遗留的登录页旧蓝光晕。

**Architecture:** 菜单结构提取为数据（`menuConfig.ts`：分组 → 子项 → 路由 → 图标 + admin 权限），rail / flyout / 面包屑三个消费者共享同一配置源。rail + flyout 封装为 `NavRail.vue`（hover 120ms 延迟预览 + 点击行为按 Q1-a：单页分组直达、多页分组开 flyout）；用户区收进 rail 底部头像（Q2-a，点击弹菜单：用户名/角色 + 退出登录）；顶栏只剩分组级面包屑（Q3-a）。登录页去"登录/注册 tabs"笔误（项目无注册功能）。

**Tech Stack:** Vue 3 `<script setup>` + TS + Element Plus（仅保留 el-tooltip/el-dropdown 等轻组件）+ `@phosphor-icons/vue`；测试 vitest（menuConfig 用 node 环境，NavRail 用 jsdom + @vue/test-utils）。

## Global Constraints

- **菜单标签/顺序/路由零改动（硬约束）**：`menuConfig.ts` 必须与现 `MainLayout.vue` 的 `el-menu` 逐项一致（7 分组、15 个子项、1 个平铺项），由不变量测试锁定；**文案与 Task 11 同罚**（M2 曾因改文案返工）
- **权限显隐保留**：系统管理分组仅 admin 可见；`v-if` 条件语义不变
- **路由与默认落地页零改动**：不新增/删除/改名任何路由；登录后默认落地 `/basic-info/category` 不变；`/m/` 移动端与 `MobileLayout.vue` 零改动
- **只改布局**：`el-main` 内的 `<router-view />` 及各页面内容零改动（M2 的页面结构不动）
- 样式只用规范 §1 令牌；图标一律 `@phosphor-icons/vue`（`Ph*`，用前核对导出名）；禁手绘 SVG；禁新增硬编码色值
- **可达性**：flyout 支持 focus-within 打开、Esc 与点击外部关闭；交互元素 focus-visible 可见；`--ease` 唯一缓动
- 提交前缀 `style(restyle-m3):` / `feat(restyle-m3):`；每 Task 独立 commit
- 分支：从 `dev` 拉 `feature/restyle-m3-shell`

import type { Component } from 'vue'
import {
  PhChartBar, PhFolderOpen, PhHandCoins, PhListChecks, PhReceipt, PhShieldCheck, PhTruck, PhWallet,
} from '@phosphor-icons/vue'

export interface MenuChild {
  path: string
  label: string
}

export interface MenuGroup {
  key: string
  label: string
  icon: Component
  /** 仅 admin 可见（现系统管理分组的显隐语义） */
  adminOnly?: boolean
  children: MenuChild[]
  /** 平铺单页项（如数据分析）的直达路由；children 非空时不用 */
  path?: string
}

export const MENU_GROUPS: MenuGroup[] = [
  {
    key: 'basic-info', label: '基础信息', icon: PhFolderOpen,
    children: [
      { path: '/basic-info/category', label: '类目' },
      { path: '/basic-info/product', label: '产品库' },
      { path: '/basic-info/factory', label: '工厂库' },
      { path: '/basic-info/logistics', label: '物流服务商' },
      { path: '/basic-info/customer', label: '客户' },
    ],
  },
  {
    key: 'system', label: '系统管理', icon: PhShieldCheck, adminOnly: true,
    children: [
      { path: '/system/approvals', label: '待审批' },
      { path: '/system/users', label: '用户管理' },
      { path: '/system/logs', label: '操作日志' },
      { path: '/system/backups', label: '备份管理' },
    ],
  },
  { key: 'orders', label: '订单管理', icon: PhReceipt, children: [{ path: '/orders/list', label: '订单列表' }] },
  { key: 'tracking', label: '跟单管理', icon: PhListChecks, children: [{ path: '/tracking', label: '跟单工作台' }] },
  {
    key: 'factory-payment', label: '工厂结算', icon: PhHandCoins,
    children: [
      { path: '/factory-payment', label: '结算单' },
      { path: '/factory-payment/statement', label: '对账单' },
    ],
  },
  { key: 'logistics', label: '物流管理', icon: PhTruck, children: [{ path: '/logistics', label: '发货单' }] },
  { key: 'finance', label: '轻财务', icon: PhWallet, children: [{ path: '/finance', label: '收支流水' }] },
  { key: 'analytics', label: '数据分析', icon: PhChartBar, children: [], path: '/analytics' },
]

/** 路由 → 所属分组（前缀匹配；/analytics 精确匹配） */
export function groupOfPath(path: string): MenuGroup | undefined {
  return MENU_GROUPS.find((g) => {
    if (g.children.length === 0) return g.path === path
    return g.children.some((c) => path === c.path || path.startsWith(c.path + '/')) || path.startsWith('/' + g.key)
  })
}

<script setup lang="ts">
import { computed, onBeforeUnmount, onMounted, ref } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { MENU_GROUPS, groupOfPath } from '../../layouts/menuConfig'
import { useUserStore } from '../../stores/user'

const route = useRoute()
const router = useRouter()
const userStore = useUserStore()

const isAdmin = computed(() => userStore.roles.includes('admin'))
const visibleGroups = computed(() => MENU_GROUPS.filter((g) => !g.adminOnly || isAdmin.value))

/** 当前路由所属分组（rail 激活态） */
const activeKey = computed(() => groupOfPath(route.path)?.key ?? '')

/** flyout 展开/悬停的分组；单页分组无 flyout */
const openKey = ref<string | null>(null)
let hoverTimer: ReturnType<typeof setTimeout> | null = null

function toggleOrGo(key: string) {
  const g = MENU_GROUPS.find((x) => x.key === key)!
  if (!g.children.length) return go(g)
  if (openKey.value === key) return closeFlyout()
  openKey.value = key
}

function hoverOpen(key: string) {
  const g = MENU_GROUPS.find((x) => x.key === key)!
  if (!g.children.length || g.children.length === 1) return
  if (hoverTimer) clearTimeout(hoverTimer)
  hoverTimer = setTimeout(() => { openKey.value = key }, 120)
}

function hoverLeave() {
  if (hoverTimer) clearTimeout(hoverTimer)
}

function go(g: (typeof MENU_GROUPS)[number]) {
  openKey.value = null
  const target = g.children.length ? g.children[0].path : g.path
  if (target && route.path !== target) router.push(target)
}

function onKeydown(e: KeyboardEvent) {
  if (e.key === 'Escape') openKey.value = null
}
function onDocClick(e: MouseEvent) {
  if (!(e.target as HTMLElement).closest('.nav-rail')) openKey.value = null
}
onMounted(() => { document.addEventListener('click', onDocClick) })
onBeforeUnmount(() => { document.removeEventListener('click', onDocClick) })

const userInitial = computed(() => (userStore.username || '?').slice(0, 1).toUpperCase())
const userPop = ref(false)
function toggleUserPop() { userPop.value = !userPop.value }
function onLogout() {
  userPop.value = false
  userStore.logout()
  router.push('/login')
}
</script>
